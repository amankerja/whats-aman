Kode Google Apps Script (.gs)

Tempelkan langsung ke Google Sheets
Code.gs JavaScript (ES6+) 
function onFormSubmit(e) {
  const webhookUrl = "https://api.whatsaman.com/v1/integrations/webhook/google_form/nomer-aman-kerja";
  
  // Mengambil field Nama dan No WhatsApp otomatis
  const nama = e.namedValues["Nama Lengkap"] ? e.namedValues["Nama Lengkap"][0] : (e.namedValues["Nama"] ? e.namedValues["Nama"][0] : "Pelanggan");
  const phone = e.namedValues["Nomor WhatsApp"] ? e.namedValues["Nomor WhatsApp"][0] : e.namedValues["No WA"][0];
  
  const payload = {
    form_name: "Formulir Pendaftaran",
    name: nama,
    phone: phone,
    data: e.namedValues
  };
  
  const options = {
    method: "post",
    contentType: "application/json",
    payload: JSON.stringify(payload)
  };
  
  UrlFetchApp.fetch(webhookUrl, options);
}
